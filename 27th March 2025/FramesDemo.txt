using System;
using System.Collections.Generic;
using System.Linq;
using System.Net.NetworkInformation;
using System.Text;
using System.Threading;
using System.Threading.Tasks;
using OpenQA.Selenium;
using OpenQA.Selenium.Chrome;
using OpenQA.Selenium.DevTools.V132.CSS;
using OpenQA.Selenium.Edge;
using OpenQA.Selenium.Firefox;
using OpenQA.Selenium.Interactions;
using OpenQA.Selenium.Support.UI;
using SeleniumExtras.WaitHelpers;


namespace SecondProject
{
    internal class FirstClass
    {

        public static void Main(string[] args)
        {
            IWebDriver driver = new ChromeDriver();
            driver.Manage().Window.Maximize();
            driver.Url = "https://letcode.in/frame";

            driver.SwitchTo().Frame("firstFr");

            driver.FindElement(By.Name("fname")).SendKeys("Arun");
            driver.FindElement(By.Name("lname")).SendKeys("Motoori");

            IWebElement childFrame = driver.FindElement(By.CssSelector("iframe[src='innerframe']"));

            driver.SwitchTo().Frame(childFrame);

            driver.FindElement(By.Name("email")).SendKeys("arun.motoori@gmail.com");
            Thread.Sleep(3000);
            driver.FindElement(By.Name("email")).Clear();
            Thread.Sleep(3000);
            driver.SwitchTo().ParentFrame();
            Thread.Sleep(3000);
            driver.FindElement(By.Name("fname")).Clear();
            Thread.Sleep(3000);
            driver.FindElement(By.Name("lname")).Clear();
            Thread.Sleep(3000);
            driver.SwitchTo().DefaultContent();
            Thread.Sleep(3000);
            string pageHeading =  driver.FindElement(By.CssSelector("h1[class='title has-text-centered is-pulled-left']")).Text;
            Console.WriteLine(pageHeading);
            Thread.Sleep(3000);
            driver.Quit();


        }


    }
}
