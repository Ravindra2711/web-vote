            driver.Manage().Window.Maximize();
            driver.Url = "https://omayo.blogspot.com/";

            Screenshot screenshot = ((ITakesScreenshot)driver).GetScreenshot();
            screenshot.SaveAsFile("C:\\Users\\arunm\\source\\repos\\SecondSolution\\SecondProject\\Screenshots\\HomePage.png");

            driver.FindElement(By.LinkText("jqueryui")).Click();

            screenshot = ((ITakesScreenshot)driver).GetScreenshot();
            screenshot.SaveAsFile("C:\\Users\\arunm\\source\\repos\\SecondSolution\\SecondProject\\Screenshots\\JQueryUIPage.png");

            driver.Quit();

        }

    }
